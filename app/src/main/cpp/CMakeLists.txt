cmake_minimum_required(VERSION 3.22.1)
project(studyforge LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS sf_compile_pool=2)
set(CMAKE_JOB_POOL_COMPILE sf_compile_pool)
option(SF_LOCAL_INFERENCE "Compile the pinned offline image runtime" ON)
set(SF_SHARED "${CMAKE_CURRENT_LIST_DIR}/../../../../shared")
if(SF_LOCAL_INFERENCE)
 include("${SF_SHARED}/runtime.cmake")
endif()
add_library(native-lib SHARED native-lib.cpp local-engine.cpp "${SF_SHARED}/sf_engine.cpp")
target_include_directories(native-lib PRIVATE "${SF_SHARED}")
target_compile_features(native-lib PRIVATE cxx_std_17)
if(SF_LOCAL_INFERENCE)
 target_compile_definitions(native-lib PRIVATE SF_LOCAL_INFERENCE=1)
 target_link_libraries(native-lib PRIVATE stable-diffusion)
endif()
target_link_options(native-lib PRIVATE "-Wl,-z,max-page-size=16384")
