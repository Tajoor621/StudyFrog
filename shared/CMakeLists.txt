cmake_minimum_required(VERSION 3.22)
project(StudyForgeNative LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS sf_compile_pool=2)
set(CMAKE_JOB_POOL_COMPILE sf_compile_pool)
include("${CMAKE_CURRENT_LIST_DIR}/runtime.cmake")
add_library(sf_engine SHARED sf_engine.cpp sf_engine.h)
target_compile_features(sf_engine PRIVATE cxx_std_17)
target_compile_definitions(sf_engine PRIVATE SF_LOCAL_INFERENCE=1)
target_link_libraries(sf_engine PRIVATE stable-diffusion)
target_include_directories(sf_engine PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
if(APPLE)
 set_target_properties(sf_engine PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A PUBLIC_HEADER sf_engine.h MACOSX_FRAMEWORK_IDENTIFIER com.scenio.mth.inference XCODE_ATTRIBUTE_DEFINES_MODULE YES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
endif()
